bit全探索（N ≤ 20）

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n;
    ll x;
    cin >> n >> x;
    vector<ll> a(n);
    for (auto &v : a) cin >> v;

    bool ok = false;
    for (int bit = 0; bit < (1 << n); bit++) {
        ll sum = 0;
        for (int i = 0; i < n; i++) {
            if (bit & (1 << i)) sum += a[i];   // i番目を選んだ
        }
        if (sum == x) ok = true;
    }
    cout << (ok ? "Yes" : "No") << "\n";
    return 0;
}

答えを二分探索（めぐる式）

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n;
    ll k;
    cin >> n >> k;
    vector<ll> a(n);
    for (auto &x : a) cin >> x;

    // x が条件を満たすか。x が大きいほど満たしやすい場合
    auto check = [&](ll x) -> bool {
        ll cnt = 0;
        for (auto v : a) cnt += (v + x - 1) / x - 1;
        return cnt <= k;
    };

    ll ng = 0, ok = 2e9;      // ng は満たさない、ok は満たす
    while (abs(ok - ng) > 1) {
        ll mid = (ok + ng) / 2;
        if (check(mid)) ok = mid;
        else ng = mid;
    }
    cout << ok << "\n";
    return 0;
}

ng と ok の初期値が正しいか必ず確認する。ng には絶対に条件を満たさない値、ok には絶対に満たす値を入れる。逆向き（小さいほど満たしやすい）なら ng と ok を入れ替える。

BFS（グリッドの最短距離）

#include <bits/stdc++.h>
using namespace std;

int main() {
    int h, w;
    cin >> h >> w;
    vector<string> s(h);
    for (auto &row : s) cin >> row;

    int dx[] = {1,-1,0,0}, dy[] = {0,0,1,-1};
    vector<vector<int>> dist(h, vector<int>(w, -1));

    queue<pair<int,int>> q;
    q.push({0, 0});
    dist[0][0] = 0;

    while (!q.empty()) {
        auto [x, y] = q.front();
        q.pop();
        for (int d = 0; d < 4; d++) {
            int nx = x + dx[d], ny = y + dy[d];
            if (nx < 0 || nx >= h || ny < 0 || ny >= w) continue;
            if (s[nx][ny] == '#') continue;
            if (dist[nx][ny] != -1) continue;
            dist[nx][ny] = dist[x][y] + 1;
            q.push({nx, ny});
        }
    }
    cout << dist[h-1][w-1] << "\n";
    return 0;
}

BFS（グラフの最短距離）

#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        a--; b--;                    // 1始まり → 0始まり
        g[a].push_back(b);
        g[b].push_back(a);           // 有向なら削る
    }

    vector<int> dist(n, -1);
    queue<int> q;
    q.push(0);
    dist[0] = 0;

    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int to : g[v]) {
            if (dist[to] != -1) continue;
            dist[to] = dist[v] + 1;
            q.push(to);
        }
    }
    for (int i = 0; i < n; i++) cout << dist[i] << "\n";
    return 0;
}

DFS（連結成分の個数）

#include <bits/stdc++.h>
using namespace std;

int h, w;
vector<string> s;
vector<vector<bool>> vis;
int dx[] = {1,-1,0,0}, dy[] = {0,0,1,-1};

void dfs(int x, int y) {
    vis[x][y] = true;
    for (int d = 0; d < 4; d++) {
        int nx = x + dx[d], ny = y + dy[d];
        if (nx < 0 || nx >= h || ny < 0 || ny >= w) continue;
        if (s[nx][ny] != '#') continue;
        if (vis[nx][ny]) continue;
        dfs(nx, ny);
    }
}

int main() {
    cin >> h >> w;
    s.resize(h);
    for (auto &row : s) cin >> row;
    vis.assign(h, vector<bool>(w, false));

    int cnt = 0;
    for (int i = 0; i < h; i++)
        for (int j = 0; j < w; j++)
            if (s[i][j] == '#' && !vis[i][j]) { dfs(i, j); cnt++; }

    cout << cnt << "\n";
    return 0;
}

DP（1次元・MODあり）

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll MOD = 998244353;

int main() {
    int n;
    cin >> n;

    // dp[i] = i 段目に到達する方法の数
    vector<ll> dp(n + 1, 0);
    dp[0] = 1;

    for (int i = 1; i <= n; i++) {
        dp[i] = (dp[i] + dp[i-1]) % MOD;
        if (i >= 2) dp[i] = (dp[i] + dp[i-2]) % MOD;
    }
    cout << dp[n] << "\n";
    return 0;
}

DP（ナップサック・2次元）

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n, W;
    cin >> n >> W;
    vector<int> w(n), v(n);
    for (int i = 0; i < n; i++) cin >> w[i] >> v[i];

    // dp[i][j] = i個目まで見て、重さ j 以下での最大価値
    vector<vector<ll>> dp(n + 1, vector<ll>(W + 1, 0));

    for (int i = 0; i < n; i++) {
        for (int j = 0; j <= W; j++) {
            dp[i+1][j] = max(dp[i+1][j], dp[i][j]);            // 選ばない
            if (j + w[i] <= W)
                dp[i+1][j+w[i]] = max(dp[i+1][j+w[i]], dp[i][j] + v[i]);
        }
    }
    cout << dp[n][W] << "\n";
    return 0;
}

Union-Find

#include <bits/stdc++.h>
using namespace std;

struct UnionFind {
    vector<int> par;
    UnionFind(int n) : par(n, -1) {}
    int root(int x) { return par[x] < 0 ? x : par[x] = root(par[x]); }
    bool same(int x, int y) { return root(x) == root(y); }
    void unite(int x, int y) {
        x = root(x); y = root(y);
        if (x == y) return;
        if (par[x] > par[y]) swap(x, y);
        par[x] += par[y];
        par[y] = x;
    }
    int size(int x) { return -par[root(x)]; }
};

int main() {
    int n, m;
    cin >> n >> m;
    UnionFind uf(n);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        a--; b--;
        uf.unite(a, b);
    }

    int best = 0;
    for (int i = 0; i < n; i++) best = max(best, uf.size(i));
    cout << best << "\n";
    return 0;
}

bool is_prime(ll n) {
    if (n < 2) return false;
    for (ll i = 2; i * i <= n; i++)

    return true;
}

vector<ll> divisors(ll n) {
    vector<ll> res;

        if (n % i == 0) {
            res.push_back(i);
            if (i != n / i) res.push_back(n / i);

    return res;
}

通せる見込みのあるものだけ。新規は開かない

問題を開いたらまず見る順番

  1. 制約 — N の上限で解法の候補が絞れる
  2. 出力 — Yes/No なのか個数なのか最小値なのか
  3. サンプル1 — 問題文が分からなくてもサンプルで分かることが多い
  4. 本文

問題文を頭から精読すると時間を溶かす。制約と出力形式を先に見ると、読むべき部分が絞れる。

ペナルティは提出ごとに5分。ただし最終的にACできなければペナルティは加算されないので、自信がなくても出す価値はある。ただしサンプルは必ず全部通してから。

逆引き：この表現が出たらこれ

問題文の言い回しから使う道具を引く。B〜D問題はここでほぼ方針が決まる。

WAが出たときのチェックリスト

上から順に確認する。B〜C問題のWAはほぼこの中にある。

型・オーバーフロー

  • int で足りているか。合計・積が 2×109 を超えないか
  • accumulate の初期値が 0 になっていないか（0LL が必要）
  • MOD を取る前に掛け算でオーバーフローしていないか
  • int同士の割り算で切り捨てられていないか

境界・添字

  • ループの条件が i < n か i <= n か
  • 入力が1始まりなのに0始まりで扱っていないか
  • 累積和のサイズが n+1 になっているか
  • N=1 のとき、K=0 のときに壊れないか

出力

  • Yes / YES / yes が問題文と一致しているか
  • 小数の桁数を指定したか（setprecision）
  • デバッグ用の cout が残っていないか
  • 複数ケースある問題で、変数を初期化し直しているか

ロジック

  • 最大値の初期値が小さすぎないか（負の数がある場合 0 だとダメ）
  • BFS の訪問済みチェックを push の時点でやっているか
  • ソートしてから lower_bound を使っているか
  • next_permutation の前にソートしたか
  • 貪欲が本当に正しいか（反例を1つ探してみる）

サンプルが全部通るのにWAなら、境界値を疑う。N=1、全部同じ値、最大値ちょうど、答えが0になるケース。この4つを手で作って試す。

TLEが出たとき

どこが重い？ └ 二重ループがある → 片方を map / set / 累積和で置き換えられないか └ ループ内で sort している → 外に出せないか └ ループ内で string を連結している → += に変える、または後でまとめて出力 └ 再帰が深い → メモ化していないなら追加 └ cout を大量に呼んでいる → endl を "\n" に変える

出力が遅いとき

ios::sync_with_stdio(false);
cin.tie(nullptr);

main の先頭に書くと入出力が速くなる。105 行以上を出力する問題では効く。ただし printf と混ぜてはいけない。

計算量の見積もり

N の値をループの回数に当てはめて、108 を超えていないか確認する。超えていたら解法を変える必要がある。定数倍の改善では間に合わない。

REが出たとき

  • 配列の範囲外にアクセスしていないか（a[n] は存在しない）
  • 0で割っていないか
  • vector のサイズを確保する前に a[i] = x していないか
  • 再帰が深すぎてスタックオーバーフローしていないか
  • 空の vector に対して a[0] や *max_element を呼んでいないか

手元では動くのにREになる場合、範囲外アクセスの可能性が高い。a[i] を a.at(i) に変えると範囲外で例外が出るので、原因の特定に使える（提出前に戻す）。

ローカルでの回し方

「サンプルを全部試す」を毎回手でやると時間を溶かす。最初に環境を作っておくと、1回の提出あたり数分変わる。

コンパイル

g++ -std=c++20 -O2 -o a main.cpp
./a

AtCoderのジャッジは C++20 / GCC。手元も揃えておくと、通るはずのものが通らない事故が減る。

サンプルを流す

入力を毎回タイプするのは非効率。ファイルに貼って流し込む。

# in.txt にサンプル入力を貼っておく
./a < in.txt

複数サンプルがあるなら in1.txt, in2.txt と分けて、順に流す。

for f in in*.txt; do echo "--- $f"; ./a < $f; done

バグを実行時に検出する

g++ -std=c++20 -fsanitize=address,undefined -g -o a main.cpp

これが一番効く。配列の範囲外アクセス、符号付き整数のオーバーフロー、未初期化の使用を、実行した瞬間に行番号つきで教えてくれる。RE や原因不明の WA の特定時間が桁違いに短くなる。

実行は遅くなるので、提出用のビルドとは分けて使う。

online-judge-tools（推奨）

サンプルの取得とテストを自動化するツール。導入すると「サンプルを全部試す」が1コマンドになる。

pip3 install online-judge-tools
# 問題ページのURLからサンプルを取得
oj d https://atcoder.jp/contests/abc300/tasks/abc300_a

# 全サンプルでテスト（差分も出る）
g++ -std=c++20 -O2 -o a main.cpp
oj t -c ./a

oj d で test/ ディレクトリにサンプルが落ちてくる。oj t は全ケースを流して、期待値と実際の出力の差分を色付きで出す。どのサンプルで落ちたかが一目で分かる。

コンテスト中に一番差が出るのはここ。手でコピペしていると1問あたり1〜2分かかる。100分で5問触れば10分の差になる。

作業ディレクトリの型

abc300/
  a/
    main.cpp
    test/        # oj d で自動生成
  b/
    main.cpp
    test/

問題ごとにディレクトリを切ると、前の問題のコードを消さずに済む。戻って直すときに効く。

提出

ブラウザに貼るのが基本だが、oj からも出せる。

oj s main.cpp

提出前に必ず oj t を通す。サンプルが通っていない状態で出すとペナルティが増えるだけ。逆に、サンプルが全部通っていれば出す価値がある。

提出後の1〜2分

  • ジャッジ待ちの間に次の問題を開いて読み始める
  • WAだった場合に備えて、コードは閉じない
  • ACが出たら、その問題のタブは閉じて視界から消す
  • ジャッジ結果を見つめて待たない。100分のうち数分が積み上がる

他人のコードから学ぶ

競プロで最も効率のいい学習手段。解説を読むより速く理解できることがある。

  1. 問題ページを開く

  3. 結果を「AC」、言語を「C++」で絞り込む

  • 自分より短く書けている部分はどこか
  • 知らない関数を使っていないか
  • 場合分けをどう減らしているか
  • 入出力の書き方（自分より速い形があるか）

C++入門

プログラミングもC++も初めての人向け。上から読めばA問題が解けるところまで行く。数学の知識は要らない。書ける人は飛ばしてよい。

5分でわかるC++

A問題を解くのに必要なC++は、これだけです。この1ページを読めば書き始められます。分からない部分があったら、下の各章で詳しく説明しています。

1. この形で始める（暗記不要、コピーでいい）

#include <bits/stdc++.h>
using namespace std;

int main() {
    // ここに書く
    return 0;
}

上2行と int main() は毎回同じです。意味は分からなくて構いません。

2. 出す（cout）

cout << 100 << endl;        // 数字
cout << "Yes" << endl;      // 文字列はクォートで囲む
cout << a << " " << b << endl; // 間にスペース

endl は改行。行末のセミコロンを忘れるとエラーになります。

3. 受け取る（cin）

int a, b;
cin >> a >> b;      // 入力が「3 5」なら a=3, b=5

string s;
cin >> s;           // 文字列

矢印の向きに注意。出すのが <<、受け取るのが >>。スペース区切りでも改行区切りでも同じように読めます。

4. 入れ物（型）

型入れるもの使うとき
int整数基本これ。±21億まで
long long大きい整数10億を超える計算をするとき
string文字列文字が来るとき
double小数小数が答えのとき

迷ったら long long にしておけば安全です。

5. 計算する

a + b     a - b     a * b     a / b
a % b     // あまり（7 % 3 は 1）

整数どうしの割り算は切り捨てられる。7 / 2 は 3。小数が欲しいなら 7.0 / 2。

6. 条件で分ける（if）

if (a > b) {
    cout << "Yes" << endl;
} else {
    cout << "No" << endl;
}
書き方意味
a == b等しい（イコール2つ）
a != b等しくない
a >= ba は b 以上
a > ba は b より大きい
A && BAかつB
A || BAまたはB

1行で書きたいときはこれも使えます。

cout << (a > b ? "Yes" : "No") << endl;

7. 繰り返す（for）

for (int i = 0; i < n; i++) {
    // n回くり返される（i は 0, 1, 2, ... n-1）
}

この形はほぼ毎回同じです。「n回やりたい」ならこう書く、と覚えれば足ります。

8. たくさん受け取る（配列）

int n;
cin >> n;

vector<int> a(n);           // n個の入れ物
for (auto &x : a) cin >> x;  // まとめて受け取る

cout << a[0] << endl;         // 1個目（番号は0から）

番号は0から始まる。n個作ったら使えるのは a[0] 〜 a[n-1]。

9. 実際に1問解いてみる

「2つの整数 A, B が与えられます。A × B を出力してください」という問題。

#include <bits/stdc++.h>
using namespace std;

int main() {
    int a, b;
    cin >> a >> b;

    cout << a * b << endl;
    return 0;
}

これで実際にACできます

実行環境は不要です。AtCoderのサイト上で完結します。

Yes / No を答える

cin >> a >> b;
cout << (a + b >= 10 ? "Yes" : "No") << endl;

for (int i = 0; i < n; i++) {
    int a;

条件を満たす個数を数える

int n, k;
cin >> n >> k;

int cnt = 0;
for (int i = 0; i < n; i++) {

最初の1問

例題

入力:

出力:
15

解く手順

  1. 入力欄を見る → 「A B」なので整数が2つ
  2. 出力欄を見る → 整数1つ
  3. 変数を2つ作って受け取る
  4. 掛けて出力する

int main() {
    int a, b;

}

実際に提出してみる

ABC169 A - Multiplication 1 がまさにこの問題です。このコードをそのまま提出すればACになります。

提出の手順：問題ページの下にあるコード入力欄に貼って「提出」を押す。数秒で結果が出ます。

1問ACできれば、あとは同じことの繰り返しです。入力を受け取って、何か計算して、出力する。難しい問題も、この形の応用でしかありません。

2章　分岐と繰り返し

条件で分ける、繰り返す。この2つでA問題はほぼ全部解ける。

if文

条件によって処理を変えたいときに使います。

int main() {
    int a;
    cin >> a;

    if (a > 10) {
        cout << "big" << endl;
    } else {
        cout << "small" << endl;
    }
    return 0;
}

aが10より大きければ「big」、そうでなければ「small」が出ます。

形

if (条件) {
    条件が正しいときの処理
} else {
    そうでないときの処理
}

3つ以上に分ける

if (a > 100) {
    cout << "large" << endl;
} else if (a > 10) {
    cout << "medium" << endl;
} else {
    cout << "small" << endl;
}

else if はいくつでも並べられます。上から順に判定して、最初に当てはまったものだけが実行されます。

else はなくてもいい

if (a > 10) {
    cout << "big" << endl;
}
// 条件を満たさなければ何もしない

1行なら中括弧を省略できる

if (a > 10) cout << "big" << endl;
else cout << "small" << endl;

省略すると、直後の1文だけがifの中身になる。2行以上書きたいときは中括弧が必要。慣れないうちは常に中括弧を書くほうが安全。

条件の書き方

比較

書き方意味
a == baとbが等しい
a != b等しくない
a < baがbより小さい
a <= baがb以下
a > baがbより大きい
a >= baがb以上

等しいかを調べるのは ==（イコール2つ）。= ひとつは「代入」という別の意味になる。if (a = 5) と書くとバグになる。ここは初心者が必ず1度は踏む。

「以上」と「より大きい」の違い

a >= 10   // 10も含む（10, 11, 12, ...）
a >  10   // 10は含まない（11, 12, ...）

問題文の「以上」「より大きい」「未満」「以下」を正確に読み取ってください。ここを間違えるとWAになります。

複数の条件をつなぐ

書き方意味
A && BAもBも正しい（かつ）
A || BAかBのどちらかが正しい（または）
!AAが正しくない（否定）
 
if (a > 0 && b > 0) {
    cout << "both positive" << endl;
}

範囲の判定

「1以上10以下」を書きたいとき。

if (1 <= a && a <= 10) { }   // 正しい

1 <= a <= 10 とは書けない。数学ではこう書くが、C++では意図しない結果になる。必ず && でつなぐ。

Yes / No を出す定番

bool ok = (a + b >= 10);
cout << (ok ? "Yes" : "No") << endl;

条件 ? Aのとき : Bのとき という書き方です。1行で書けるので競プロでよく使います。

Yes / No の綴りは問題文からコピペする。YES と Yes は別物として採点される。A問題のWAで一番多い原因がこれ。

for文

同じ処理を繰り返すときに使います。

for (int i = 0; i < 5; i++) {
    cout << i << endl;
}
0 1 2 3 4

3つの部分

for (int i = 0; i < 5; i++) {
      ↑          ↑       ↑
      最初       続ける条件  毎回やること
  1. int i = 0 — iを0から始める
  2. i < 5 — iが5未満の間は繰り返す
  3. i++ — 1周ごとにiを1増やす

結果として、iは 0, 1, 2, 3, 4 と変化して、5になったところで終わります。

この形はほぼ毎回同じです。「n回繰り返したい」なら for (int i = 0; i < n; i++) と書く。それだけ覚えれば十分です。

よく使う形

// n回繰り返す（iは 0 〜 n-1）
for (int i = 0; i < n; i++) { }

// 1 〜 n
for (int i = 1; i <= n; i++) { }

// 逆順（n-1 〜 0）
for (int i = n - 1; i >= 0; i--) { }

ループの中で入力を受け取る

「N個の数が与えられる」タイプの問題で使います。

int n;
cin >> n;

long long sum = 0;
for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    sum += a;      // 合計に足していく
}
cout << sum << endl;

入力が「3」「10 20 30」なら、60 が出ます。

やってみる

上のコードを、合計ではなく「一番大きい数」を出すように変えてみてください。ヒント：変数 mx を用意して、if (a > mx) mx = a; と書く。

ループ変数のスコープ

for (int i = 0; i < 5; i++) {
    // ここでは i が使える
}
cout << i;   // エラー。ループの外では使えない

ループの外でも使いたいなら、ループの前に宣言します。

while文

回数が決まっていない繰り返しに使います。

int n = 100;
int cnt = 0;

while (n > 1) {
    n = n / 2;
    cnt++;
}
cout << cnt << endl;   // 100を1になるまで半分にした回数

形

while (条件) {
    条件が正しい間、ずっと繰り返す
}

中で条件を変えないと無限ループになる。上の例なら n = n / 2 がないと、n > 1 がずっと成り立ち続けて止まらない。プログラムが終わらなくなったらこれを疑う。

for と while の使い分け

使うとき
for「N回繰り返す」と回数が決まっている
while「条件を満たす間ずっと」で回数が分からない

競プロでは for のほうが圧倒的に多いです。while は「割り切れる間ずっと割る」のような場面で使います。

break と continue

break — ループを抜ける

for (int i = 0; i < 10; i++) {
    if (i == 5) break;   // 5になったら抜ける
    cout << i << endl;
}
0 1 2 3 4

「条件を満たすものが見つかったら、そこで探すのをやめる」という場面で使います。

continue — 次の周回へ飛ぶ

for (int i = 0; i < 5; i++) {
    if (i == 2) continue;   // 2のときだけスキップ
    cout << i << endl;
}
0 1 3 4

見つかったかを記録する定番

bool found = false;
for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    if (a == 100) found = true;
}
cout << (found ? "Yes" : "No") << endl;

この形はよく使います。「条件を満たすものが1つでもあるか」を調べるパターンです。

入力を全部読み切る必要がある場合は break しない。途中で抜けると残りの入力が読まれないまま。ただし読まなくても問題ないことが多いので、A問題では気にしなくていい。

3章　まとめて扱う

たくさんの数や文字をまとめて扱う。B問題に必要。

配列（vector）

数をたくさん保存しておきたいとき、変数を100個作るわけにはいきません。そこで配列を使います。

vector<int> a(5);   // int が5個入る箱を作る

a[0] = 10;
a[1] = 20;

cout << a[0] << endl;   // 10

書き方

vector<int> a(5);
       ↑     ↑  ↑
       型   名前 個数

番号は0から始まる。5個作ったら使えるのは a[0] 〜 a[4]。a[5] は存在しないのでエラーになる。ここは最初つまずくところ。

作り方いろいろ

vector<int> a(5);          // 5個、全部0
vector<int> a(5, -1);      // 5個、全部-1
vector<int> a = {1, 2, 3};  // 中身を直接指定
vector<long long> a(n);    // long long でも作れる
vector<string> s(n);        // 文字列でも

よく使う操作

a.size()        // 個数を調べる
a.push_back(7); // 末尾に追加する

入力を配列に受け取る

これが最頻出の形です。

int n;
cin >> n;

vector<int> a(n);
for (int i = 0; i < n; i++) {
    cin >> a[i];
}

短く書くこともできます。

vector<int> a(n);
for (auto &x : a) cin >> x;

auto は「型は自動で決めて」という意味、& は「中身を直接いじる」という意味です。この形を丸ごと覚えてしまえば大丈夫です。

文字列

文字列も配列と似た扱いができます。

string s = "hello";

cout << s.size() << endl;   // 5（長さ）
cout << s[0] << endl;       // h（1文字目）
cout << s[1] << endl;       // e

よく使う操作

やりたいこと書き方
長さを調べるs.size()
i文字目を見るs[i]
つなげるs + "world"
末尾に足すs += "!"
比較するs == "hello"
反転するreverse(s.begin(), s.end());
一部を取り出すs.substr(1, 3)（1文字目から3文字）

文字列の比較は == でいい。他の言語だと専用の関数が必要なことがあるが、C++のstringはそのまま比較できる。

1文字ずつ調べる

string s;
cin >> s;

int cnt = 0;
for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'a') cnt++;
}
cout << cnt << endl;   // aの個数

短く書くならこう。

for (char c : s) {
    if (c == 'a') cnt++;
}

文字の判定

char c = s[0];

if (islower(c)) { }   // 小文字か
if (isupper(c)) { }   // 大文字か
if (isdigit(c)) { }   // 数字か

大文字小文字の変換

s[0] = toupper(s[0]);   // 1文字目を大文字に
s[0] = tolower(s[0]);   // 小文字に

文字の数字を数値にする

'7'（文字）と 7（数値）は別物です。変換にはこう書きます。

char c = '7';
int n = c - '0';   // 7

理由は文字コードの話になりますが、この形を覚えるだけで十分です。桁ごとに数値として扱いたいときに使います。

配列とループ

配列とループを組み合わせると、大量のデータを処理できます。

合計を求める

long long sum = 0;
for (int i = 0; i < n; i++) {
    sum += a[i];
}

最大値を求める

int mx = a[0];               // 最初の要素で初期化
for (int i = 1; i < n; i++) {
    if (a[i] > mx) mx = a[i];
}

最大値の初期値を0にしない。負の数が含まれる問題だと、0が残ってしまって間違える。a[0] か、十分小さい値で初期化する。

条件を満たす個数を数える

int cnt = 0;
for (int i = 0; i < n; i++) {
    if (a[i] % 2 == 0) cnt++;   // 偶数の個数
}

組み込みの関数を使う

実は、合計や最大値は自分で書かなくても関数があります。

*max_element(a.begin(), a.end())        // 最大値
*min_element(a.begin(), a.end())        // 最小値
accumulate(a.begin(), a.end(), 0LL)     // 合計

a.begin() と a.end() は「配列全体」を表す決まり文句です。意味が分からなくても、この形をコピーして使えます。

最大値・最小値の前に * を付ける。忘れるとエラーになる。理由（イテレータの中身を取り出す）は今は気にしなくていい。

2重ループ

ループの中にループを書けます。全部の組み合わせを試したいときに使います。

for (int i = 0; i < 3; i++) {
    for (int j = 0; j < 3; j++) {
        cout << i << " " << j << endl;
    }
}
0 0 0 1 0 2 1 0 1 1 1 2 2 0 2 1 2 2

外側が1周する間に、内側が3周します。合計で 3 × 3 = 9回実行されます。

全ペアを調べる

「2つ選ぶ組み合わせ」を全部試すときの定番です。

for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
        // a[i] と a[j] のペア
    }
}

内側を j = i + 1 から始める。j = 0 だと (1,2) と (2,1) を両方数えてしまい、答えが2倍になる。

回数に注意

2重ループは、Nが大きいと一気に遅くなります。

200000400億回絶対に間に合わない

計算量を言葉で言うと

「O(N²)」のような書き方は難しそうに見えますが、言い換えるとこうです。

よく出る算数

余り（あまり）

割り算のあまりのこと。% で計算できます。

7 % 3   // 7を3で割ったあまり = 1
10 % 2  // 0（割り切れる）

「偶数かどうか」は x % 2 == 0 で調べられます。2で割ってあまりが0なら偶数です。

切り上げ

「7個を3個ずつの袋に入れると何袋いる?」→ 3袋（2袋では足りないので切り上げ）。

7 / 3        // 2（切り捨てられる）

切り上げの式は覚えるだけでいいです。理由を理解しなくても使えます。

1からNまでの合計

1+2+3+...+N を計算したいとき。for文で足してもいいですが、式でも出せます。

N * (N + 1) / 2

Nが大きいとfor文では間に合わないことがあるので、そのときだけこの式を使います。

N個から2個選ぶ組み合わせの数

N * (N - 1) / 2

「5人から2人選ぶ組は何通り?」なら 5 × 4 ÷ 2 = 10通り。これも覚えるだけで大丈夫です。

この2つの式は int だとあふれる。Nが20万だと答えが200億くらいになるので、long long を使う。

A問題 完全攻略

A問題は難しさではなく速さ。型を全部並べて、読む→書く→確認を固める。

A問題を3分で解く手順

時間配分

時間やること
0:00〜0:30入力形式と出力形式だけ見る。本文は後
0:30〜1:00サンプル1で「何をすればいいか」を掴む
1:00〜2:00テンプレを貼って書く

読む順番を固定する

  1. 入力欄 — 何が何個来るか。これでコードの前半が決まる
  2. 出力欄 — 整数1つか、Yes/Noか、文字列か
  3. サンプル1 — 入力と出力の対応を見る。これで大体分かる
  4. 本文 — 上で分からなかったときだけ読む

A問題の本文は状況説明が長いことが多く、頭から読むと時間を溶かす。入力・出力・サンプルの3点だけで解けることが大半。

A問題の型（全14種）

ABCのA問題は、ほぼこの14種のどれかに当てはまる。型が分かれば書く内容が即決まる。

型1：四則演算だけ

与えられた数を計算して出すだけ。最も多い型。

int a, b;
cin >> a >> b;
cout << a * b << endl;

注意：割り算が出たら整数か小数かを確認する。7/2 は 3 になる。

型2：条件分岐でYes/No

条件を満たすかどうかを判定する。

int a, b, c;
cin >> a >> b >> c;
bool ok = (a + b + c >= 180) && (a >= 40);
cout << (ok ? "Yes" : "No") << endl;

注意：出力文字列は問題文からコピペする。YES と Yes は別物。

型3：偶数・奇数の判定

int a, b;
cin >> a >> b;
if (a * b % 2 == 0) cout << "Even" << endl;
else cout << "Odd" << endl;

注意：積の偶奇は「どちらか一方でも偶数なら偶数」。掛けてから判定しても、片方ずつ見てもいい。

型4：3つの数を比較する

最大・最小・真ん中を求める。場合分けよりソートが速い。

vector<int> v(3);
for (auto &x : v) cin >> x;
sort(v.begin(), v.end());
cout << v[1] << endl;   // 真ん中

注意：「2番目に大きい」と「2番目に小さい」を取り違えない。

型5：文字列の1文字を見る／変える

string s;
cin >> s;
s[0] = toupper(s[0]);
cout << s << endl;

注意：s[0] は char。string と比較するとエラーになる。

型6：文字列の中を数える

string s;
cin >> s;

int cnt = 0;
for (char c : s) if (c == '1') cnt++;
cout << cnt << endl;

注意：C++のstringは == で比較できる。strcmp は不要。

A問題で落とす原因

A問題のWAは実力ではなく事故。原因はほぼこの中にある。提出前に確認する。

出力まわり（最多）

  • Yes / YES / yes を問題文と照合したか
  • Even / Odd、Positive / Negative なども同様
  • 改行を入れたか
  • 複数出力のとき、区切りがスペースか改行か確認したか
  • 小数の桁数を指定したか

計算まわり

  • 整数の割り算で切り捨てられていないか
  • 切り上げのつもりで切り捨てになっていないか
  • int で足りているか（積が109を超えないか）
  • マイナスの余りが出る計算をしていないか

読み違い

  • 「以上」と「より大きい」を取り違えていないか
  • 「2番目に大きい」を「2番目に小さい」と読んでいないか
  • 入力の順番が問題文どおりか（H W なのに W H で読んでいないか）
  • 1始まりの番号を答えるのに0始まりで出していないか

A問題でWAが出たら、まず出力文字列を疑う。ロジックを見直す前に、Yes/No の綴りと大文字小文字を問題文と1文字ずつ照合する。ここが原因のことが本当に多い。

ドリルのやり方

  2. 入力・出力・サンプル1だけ読む

現在の時刻が H 時 M 分です。K 分前の時刻を出力してください。日をまたぐ場合は前日の時刻を出力します。

単位を揃える — 時と分を別々に計算すると繰り下がりでバグる。全部「分」に直してから計算し、最後に戻す。

日をまたぐ — マイナスになったら1日分（1440分）を足す。

#include <bits/stdc++.h>
using namespace std;

int main() {
    int h, m, k;
    cin >> h >> m >> k;

    int total = h * 60 + m;      // 全部を分に直す
    total -= k;
    if (total < 0) total += 24 * 60;   // 日をまたいだ

    cout << total / 60 << " " << total % 60 << endl;
    return 0;
}

時刻の問題は「分に統一」で必ず解けます。時と分を別々に扱おうとすると場合分けが増えて事故ります。単位を1つに揃えるのが定石です。

#include <bits/stdc++.h>

int main() {

    bool ok = false;
    for (int a = 0; a * 4 <= n; a++) {
        for (int b = 0; a * 4 + b * 7 <= n; b++) {

    cout << (ok ? "Yes" : "No") << endl;

}

A - 料金区分A

この型が一番よく出る。境界値をどちらに含めるかを問題文で確認する。「12歳以下」なら <= 12。

上から順に判定する — else if は最初に当てはまったものだけ実行されるので、範囲の重複を気にしなくていい。

#include <bits/stdc++.h>
using namespace std;

int main() {
    int x;
    cin >> x;

    if (x <= 12) cout << "child" << endl;
    else if (x <= 64) cout << "adult" << endl;
    else cout << "senior" << endl;

    return 0;
}

制約が小さいので計算量は気にしない。詰まるのは読み違いだけ。

1以上3以下の整数 X が与えられます。X と異なる1以上3以下の整数を出力してください。

制約
X は 1以上3以下の整数
入力
X
出力

入力例
1
出力例
2

制約 — X は1〜3の3通りしかない。計算量を考える必要は一切ない。int で足りる。

ここが特徴 — 「複数ある場合はどれでもよい」と書いてある。答えが一意でない問題。X=1 なら 2 でも 3 でも正解になる。

この一文が効く

「どれでもよい」と書いてあったら、一番書きやすいものを選んでいい。最適な答えを探す必要はありません。条件さえ満たしていればACです。

解き方1：場合分けする（一番素直）

Xが1なら2、Xが2なら1、Xが3なら1。何を返してもいいので、適当に決めます。

#include <bits/stdc++.h>
using namespace std;

int main() {
    int x;
    cin >> x;

    if (x == 1) cout << 2 << endl;
    else cout << 1 << endl;

    return 0;
}

X が1なら2、それ以外なら1。これで全ケース通ります。3通りしかないので、場合分けで十分です。

解き方2：全探索する（思考停止で書ける）

1から3まで全部試して、Xと違うものが見つかったら出して終わり。

#include <bits/stdc++.h>
using namespace std;

int main() {
    int x;
    cin >> x;

    for (int i = 1; i <= 3; i++) {
        if (i != x) {
            cout << i << endl;
            return 0;      // 1つ見つけたら終了
        }
    }
    return 0;
}

候補が少ないときは全探索が速い。場合分けを考えるより、条件を満たすものを探すほうが手が早く動くことがある。ここでは3個しか候補がないので、全部試しても一瞬。

解き方3：計算で出す

1と2を入れ替えて、3のときは1を返す形。

int x;
cin >> x;
cout << (x == 1 ? 2 : 1) << endl;

三項演算子で1行。最短ですが、読みやすさは解き方1と同じです。

どれを選ぶべきか

解き方行数向いている場面
場合分け2行候補が少なく、対応が明確なとき
全探索6行条件が複雑で、場合分けを考えるのが面倒なとき
三項演算子1行条件が1つのとき

A問題は速さが命なので、思いついたものを書くのが正解です。どれが「きれい」かを考える時間があったら提出してください。

この問題から持ち帰ること

  • 「どれでもよい」を見落とさない — 出力欄に書いてあることがある。一意な答えを探す必要がなくなる
  • 候補が少ないなら全探索 — 場合分けを考えるより速いことがある
  • 見つけたら return 0 で抜ける — ループを最後まで回す必要はない
  • 制約が小さいときは計算量を考えない — X が1〜3しかないので、何をしても間に合う

この問題を3分以内に解けなかった場合、原因はほぼ「問題文の読み方」です。入力欄・出力欄・サンプルの3点を先に見る手順を試してみてください。本文の状況説明を頭から読むと、時間を使います。

「最大値を最小化」 — 逆引き表より答えを二分探索。この言い回しはほぼ確定でこれ。

考え方の転換 — 答えを直接求めるのではなく、「長さ x 以下にできるか？」を判定する関数を作る。x が大きいほど達成しやすいので、単調性がある。

判定関数 — 長さ a の丸太を x 以下にするには ceil(a/x) - 1 回切る必要がある。その合計が K 以下なら達成可能。

int n;
long long k;
cin >> n >> k;
vector<long long> a(n);
for (auto &x : a) cin >> x;

auto check = [&](long long x) {
    long long cuts = 0;
    for (auto v : a) {
        cuts += (v + x - 1) / x - 1;   // 切り上げ - 1
    }
    return cuts <= k;
};

long long lo = 0, hi = 1e9;   // lo は達成不可、hi は達成可能

    long long mid = (lo + hi) / 2;
    if (check(mid)) hi = mid;
    else lo = mid;
}

int h, w;

int h, w;
vector<string> s;
vector<vector<bool>> vis;

void dfs(int x, int y) {
    vis[x][y] = true;
    for (int d = 0; d < 4; d++) {
        int nx = x + dx[d], ny = y + dy[d];
        if (nx < 0 || nx >= h || ny < 0 || ny >= w) continue;
        if (s[nx][ny] != '#') continue;
        if (vis[nx][ny]) continue;
        dfs(nx, ny);

    s.resize(h);

                dfs(i, j);
                cnt++;
            }
        }
    }

}

const long long MOD = 1000000007;

この帯で詰まる原因

難易度が低いのに解けないとき、原因はアルゴリズムではなく実装にある。

  • 問題文の条件を読み落とす — 「以上」と「より大きい」、「連続」と「合計」など
  • ループの範囲を間違える — i < n か i <= n か
  • 1始まりと0始まりの混同 — 入力が1始まりで配列が0始まり
  • int であふれる — 合計や積が大きくなる問題
  • 実装が思いつかない — 方針は分かるが書き方が出てこない

最後の項目が一番多い。これは貼れるコードから始めて書き換えると解決する。

この帯の埋め方

  1. 上のリストを diff の小さい順に解く
  2. 20分考えて方針が立たなければ解説を見る
  3. 解説を読んだら、その日のうちに自力で書き直す
  4. 1日2〜3問。まとめてやろうとしない

40問ほど埋めると、B問題が安定して解けるようになる。そこまで来たら diff 400〜799（茶diff）に進む。

難易度は目安であって、絶対的なものではない。diff 50 の問題が解けなくて diff 300 が解けることもある。得意な型と苦手な型があるだけなので、詰まった問題は飛ばして先に進んでいい。

データ構造とアルゴリズム

B問題以降で必要になる道具。使い方と、いつ使うか。

計算量

B問題以降で最初に意識するのがこれ。AtCoderの制限時間は基本2秒で、その間に処理できるのはおおよそ108回（1億回）の計算まで。

制約は解法のヒント。N ≤ 20 なら bit全探索、N ≤ 8 なら順列全探索を疑う。問題文を読んだらまず制約を見る癖をつける。

ループの回数で考える

for (int i = 0; i < n; i++)            // N回

for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)        // N×N回

N が 105 のとき、二重ループは 1010 回になって間に合わない。この場合は別の解法を探す。

pair と tuple

2つの値をセットで扱いたいときに使う。

pair<int, int> p = {3, 5};

p.first    // 3
p.second   // 5

// vectorに入れる
vector<pair<int, int>> v;
v.push_back({1, 2});
v.push_back(make_pair(3, 4));   // これでもいい

型が違ってもいい

pair<string, int> p = {"apple", 100};

3つ以上なら tuple

tuple<int, int, int> t = {1, 2, 3};

get<0>(t)   // 1
get<1>(t)   // 2

使いにくいので、3要素以上なら構造体か vector を使う方が楽なことも多い。

ソートの応用

pairのソート

vector<pair<int, int>> v = {{3,1}, {1,2}, {3,0}};
sort(v.begin(), v.end());
// {1,2}, {3,0}, {3,1} の順

first で比較して、同じなら second で比較される。これが自動で効くので、優先順位のあるソートは pair にすると楽。

値と添字をセットでソート

vector<pair<int, int>> v(n);
for (int i = 0; i < n; i++) {
    v[i] = {a[i], i};   // {値, 元の位置}
}
sort(v.begin(), v.end());
// ソート後も元の位置が分かる

「大きい順に並べたとき何番目だったか」を聞かれる問題で使う。

自分で比較方法を決める

bool cmp(pair<int, int> a, pair<int, int> b) {
    return a.second < b.second;   // secondで昇順
}

sort(v.begin(), v.end(), cmp);

ラムダ式で書く

sort(v.begin(), v.end(), [](auto a, auto b) {
    return a.second < b.second;
});

関数を別に定義しなくていいので、こっちの方が短い。auto は型を推論してくれるキーワード。

比較関数は「aがbより前に来るか」を返す。< なら昇順、> なら降順。<= と書くとエラーになるので使わない。

map（連想配列）

キーと値を対応させて保持する。JavaScriptのオブジェクトやMapに近い。O(log N)

map<string, int> m;

m["apple"] = 100;
m["banana"] = 200;

cout << m["apple"];   // 100

存在しないキーを参照すると0で作られる。m["orange"] と書いた時点で、値0のエントリが自動的に追加される。存在チェックには count を使う。

存在確認

if (m.count("apple")) {
    // キーが存在する
}

m.erase("apple");   // 削除
m.size()             // 要素数

出現回数を数える（最頻出の使い方）

map<int, int> cnt;
for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    cnt[x]++;   // 存在しなければ0から始まるので、そのまま++できる
}

「同じ数が何個あるか」「一番多く出る値は何か」といった問題はこれで解ける。

全要素を回す

for (auto p : m) {
    cout << p.first << " " << p.second << endl;
}

// C++17以降ならこう書ける
for (auto [key, value] : m) {
    cout << key << " " << value << endl;
}

mapはキーの昇順で自動的に並ぶ。順序が要らないなら unordered_map の方が速い（O(1)）。

set（重複なしの集合）

同じ値を1つしか持たない。自動で昇順に並ぶ。O(log N)

set<int> s;

s.insert(5);
s.insert(3);
s.insert(5);   // 既にあるので追加されない

s.size()       // 2
s.count(3)     // 1（存在する）
s.count(9)     // 0（存在しない）
s.erase(3);    // 削除

種類数を数える

set<int> s;
for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    s.insert(x);
}
cout << s.size() << endl;   // 何種類あるか

最小値・最大値

*s.begin()          // 最小値
*s.rbegin()         // 最大値

重複を許すなら multiset

multiset<int> ms;
ms.insert(5);
ms.insert(5);   // 2個入る
ms.erase(ms.find(5));   // 1個だけ削除（erase(5)だと全部消える）

bit全探索

N個のものについて「選ぶ / 選ばない」の全パターンを試す。N ≤ 20 くらいまで使える。O(2N × N)

for (int bit = 0; bit < (1 << n); bit++) {
    // bit の各桁が「選ぶかどうか」を表す
    for (int i = 0; i < n; i++) {
        if (bit & (1 << i)) {
            // i番目を選んだ場合の処理
        }
    }
}

意味

1 << n        // 2のn乗
bit & (1<<i)  // bitのi桁目が1かどうか

n=3 なら bit は 0〜7 を動き、それぞれが 000, 001, 010, ... という選び方に対応する。

使用例：合計がXになる組み合わせがあるか

bool found = false;
for (int bit = 0; bit < (1 << n); bit++) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        if (bit & (1 << i)) sum += a[i];
    }
    if (sum == x) found = true;
}

順列全探索

N個の並べ方を全部試す。N ≤ 8 くらいまで。O(N!)

vector<int> p(n);
for (int i = 0; i < n; i++) p[i] = i;

do {
    // p が並び順を表す。ここで処理
} while (next_permutation(p.begin(), p.end()));

next_permutation は昇順に並んだ状態から始める必要がある。そうでないと途中のパターンからしか試されない。sort してから使う。

使用例：巡回する最短経路

vector<int> p(n);
for (int i = 0; i < n; i++) p[i] = i;

int best = 1e9;
do {
    int total = 0;
    for (int i = 0; i < n - 1; i++) {
        total += dist[p[i]][p[i+1]];
    }
    best = min(best, total);
} while (next_permutation(p.begin(), p.end()));

累積和

「区間の合計」を何度も聞かれるときに使う。前計算しておけば、各質問に O(1) で答えられる。

vector<long long> s(n + 1, 0);
for (int i = 0; i < n; i++) {
    s[i + 1] = s[i] + a[i];
}

// a[l] から a[r-1] までの合計
long long sum = s[r] - s[l];

累積和のサイズは n+1 にする。s[0] = 0 にしておくと、区間の計算で場合分けが要らなくなる。

なぜ速いか

毎回ループで足すと1回あたり O(N)。質問がQ個あると O(NQ) になる。累積和なら前計算 O(N) + 各質問 O(1) で、合計 O(N+Q) で済む。

使用例：区間の合計を Q 回聞かれる

int q;
cin >> q;
while (q--) {
    int l, r;
    cin >> l >> r;
    cout << s[r] - s[l - 1] << endl;   // 1-indexedの場合
}

二分探索

ソート済みの配列から目的の値を高速に探す。O(log N)

lower_bound / upper_bound

sort(a.begin(), a.end());   // 必ずソートしてから

// x 以上の最初の位置
int i = lower_bound(a.begin(), a.end(), x) - a.begin();

// x より大きい最初の位置
int j = upper_bound(a.begin(), a.end(), x) - a.begin();

- a.begin() を付けると添字（整数）になる。付けないとイテレータのまま。

存在確認

if (binary_search(a.begin(), a.end(), x)) {
    // x が存在する
}

個数を数える

int cnt = upper_bound(a.begin(), a.end(), x) - lower_bound(a.begin(), a.end(), x);

答えを二分探索する

「条件を満たす最小値を求めよ」という問題で、値そのものを二分探索することがある。

long long lo = 0, hi = 1e18;
while (hi - lo > 1) {

}

Union-Find

「これとこれは繋がっているか」を高速に判定する。コピペで使える定型。

struct UnionFind {
    vector<int> par;
    UnionFind(int n) : par(n, -1) {}

    int root(int x) {
        if (par[x] < 0) return x;
        return par[x] = root(par[x]);
    }
    bool same(int x, int y) {
        return root(x) == root(y);
    }
    void unite(int x, int y) {
        x = root(x); y = root(y);
        if (x == y) return;
        if (par[x] > par[y]) swap(x, y);
        par[x] += par[y];
        par[y] = x;
    }
    int size(int x) {
        return -par[root(x)];
    }
};

使い方

UnionFind uf(n);

uf.unite(0, 1);       // 0と1を繋ぐ
uf.same(0, 1)         // true
uf.size(0)            // 0が属するグループの大きさ

アルゴリズムを理解する

なぜそう書くのか、なぜ速いのか。本番中ではなく練習のときに読む。

計算量とは何を数えているか

計算量は「入力サイズが増えたとき、実行時間がどう増えるか」を表す。実際の秒数ではなく、増え方の形だけを見る。

なぜ増え方だけ見るのか

マシンの速度や定数倍は環境で変わるが、増え方は変わらない。N が10倍になったとき実行時間が10倍になるのか100倍になるのかは、アルゴリズムの性質として決まっている。ここだけ見れば「間に合うか」が判断できる。

計算量N=105 のとき意味
O(1)1回入力サイズに関係なく一定
O(log N)約17回半分ずつ減らす。二分探索
O(N)105 回全部を1回ずつ見る
O(N log N)約1.7×106 回ソート
O(N2)1010 回全ペア。この時点で間に合わない
O(2N)天文学的N が小さいときだけ

2秒で処理できるのは約108回。この一線を覚えておけば、制約を見た瞬間に「その解法で間に合うか」が判定できる。

log N が17になる理由

log は「何回半分にすれば1になるか」。105 を半分にし続けると、10万 → 5万 → 2.5万 → ... → 1 で約17回。だから二分探索は、100万個から探しても20回程度で終わる。

この「半分ずつ減らす」という発想が、二分探索・分割統治・平衡二分木（map, set）に共通する土台になっている。

定数倍は無視するが、無視できないこともある

O(N) と O(3N) は同じ O(N) だが、実際の実行時間は3倍違う。N = 107 あたりから、この差が TLE と AC を分けることがある。ただし C問題までなら、まず計算量の桁を合わせることを優先していい。

全探索の考え方

競プロの解法の多くは、突き詰めると「賢い全探索」になる。まず愚直な全探索を考えて、そこから削っていくのが基本の思考順序。

手順

  1. すべての候補を列挙する方法を考える
  2. その候補数を数える
  3. 108 以下なら、そのまま書く
  4. 超えるなら「何を全探索して、残りをどう計算で出すか」を考える

4番目が最重要

例えば「3つの数 a, b, c を選んで a+b+c = X にする」という問題を考える。素直に三重ループだと O(N3)。しかし a と b を決めれば c は X - a - b で確定するので、二重ループ＋探索で O(N2 log N) に落ちる。

これが「全探索の次元を1つ落とす」という発想で、C問題で頻出する。

// 悪い: O(N^3)
for (int i = 0; i < n; i++)
  for (int j = 0; j < n; j++)
    for (int k = 0; k < n; k++)
      if (a[i] + a[j] + a[k] == x) ok = true;

// 良い: O(N^2 log N)  3つ目は計算で出して二分探索
sort(a.begin(), a.end());
for (int i = 0; i < n; i++)
  for (int j = 0; j < n; j++) {
    int need = x - a[i] - a[j];
    if (binary_search(a.begin(), a.end(), need)) ok = true;
  }

bit全探索がなぜ「選ぶ／選ばない」に対応するか

N個の要素それぞれについて「選ぶ／選ばない」の2択があるので、パターン数は 2N。これは N ビットの2進数が取りうる値の個数と完全に一致する。

だから bit を 0 から 2N−1 まで動かせば、すべての選び方を1回ずつ列挙できる。

n = 3 のとき
bit = 0 → 000 → 何も選ばない
bit = 1 → 001 → 0番目だけ
bit = 2 → 010 → 1番目だけ
bit = 3 → 011 → 0番目と1番目
...
bit = 7 → 111 → 全部

bit & (1 << i) は「bit の i 桁目が 1 か」を調べている。1 << i は i 桁目だけが1の数（マスク）で、& を取ると該当桁以外が0になる。結果が0でなければ、その桁は立っている。

累積和：なぜ速くなるのか

区間の和を毎回ループで足すと、1回あたり O(N)。質問が Q 個なら O(NQ)。

累積和は「最初から i 番目までの合計」を先に全部作っておく。すると任意の区間の和は引き算1回で出る。

a    = [3, 1, 4, 1, 5]
s    = [0, 3, 4, 8, 9, 14]
       ↑  s[i] = 最初のi個の合計

区間 [2, 4] （1始まり）の和 = 1 + 4 + 1 = 6
                            = s[4] - s[1] = 9 - 3 = 6

なぜ s のサイズが n+1 なのか

s[0] = 0（0個の合計は0）を用意しておくと、区間の左端が先頭のときも同じ式で計算できる。これがないと「l == 1 のときだけ特別扱い」という分岐が必要になり、そこがバグの温床になる。

この発想の一般化

「前処理してから質問に答える」という構造は、累積和以外にも共通する。

  • 累積和 — 区間の和
  • 累積 max / min — 区間の最大値（区間が固定なら）
  • いもす法 — 区間への加算をまとめて処理する
  • ソート＋二分探索 — 「X 以上の個数」を何度も聞かれる場合

質問が複数回あるタイプの問題を見たら、まず「前処理で O(1) にできないか」を考える癖をつける。

2次元累積和

グリッドの矩形領域の和を求めるときに使う。1次元と同じ発想を縦横に適用する。

vector<vector<ll>> s(h+1, vector<ll>(w+1, 0));
for (int i = 0; i < h; i++)
  for (int j = 0; j < w; j++)
    s[i+1][j+1] = s[i][j+1] + s[i+1][j] - s[i][j] + a[i][j];

// 矩形 [y1,y2) x [x1,x2) の和
ll sum = s[y2][x2] - s[y1][x2] - s[y2][x1] + s[y1][x1];

足しすぎた分を引く（包除）構造になっている。s[i][j] を2回足しているので1回引く。

二分探索：2つの使い方

使い方1：ソート済み配列から探す

真ん中を見て、探している値より大きいか小さいかで半分を捨てる。これを繰り返すので O(log N)。

C++ では自分で書く必要はなく、lower_bound と upper_bound がある。

a = [1, 3, 3, 5, 7]

lower_bound(a, 3) → 位置1（3が最初に現れる位置）
upper_bound(a, 3) → 位置3（3より大きい値が最初に現れる位置）

3の個数 = upper - lower = 2

ソートされていないと嘘の答えを返す。エラーは出ない。使う前に必ず sort。

使い方2：答えそのものを二分探索する

こちらがC〜D問題で重要。「最大値を最小化」のような問題で使う。

発想の転換：答えを直接求めるのは難しいが、「答えが x 以下にできるか？」という判定問題なら簡単に解けることがある。

そして、この判定には単調性がある。x が大きいほど達成しやすい。つまり：

x:      1  2  3  4  5  6  7  8
達成:   ×  ×  ×  ×  ○  ○  ○  ○
                    ↑ここが答え

×と○の境目を二分探索で見つける。これが「答えを二分探索する」の意味。

めぐる式の書き方

ll ng = 0, ok = 2e9;      // ng は絶対ダメ、ok は絶対OK
while (abs(ok - ng) > 1) {
    ll mid = (ok + ng) / 2;
    if (check(mid)) ok = mid;
    else ng = mid;
}
// ループ後、ok が「条件を満たす最小の値」

変数名を lo/hi ではなく ng/ok にするのがコツ。どちらが答え側かを迷わなくなる。

初期値の妥当性を必ず確認する。ng には絶対に条件を満たさない値、ok には絶対に満たす値を入れる。ここが間違っていると答えがずれる。

逆向き（小さいほど満たしやすい）の問題なら、ng と ok の役割を入れ替える。

単調性の確認

二分探索が使えるのは、判定結果が「×××○○○」の形になるときだけ。「×○×○」のように交互になる問題では使えない。

問題を見たら「x を大きくしたとき、達成しやすくなる一方か？」を確認する。これが成り立てば二分探索が使える。

BFS と DFS：何が違うのか

探索の順序が違う

BFS は近いところから順に広げる。DFS は行けるところまで進んでから戻る。

スタートから見た訪問順

BFS: 距離1のマス全部 → 距離2のマス全部 → ...
     （波紋が広がるように）

DFS: 一方向に進めるだけ進む → 行き止まりで戻る → 別方向へ
     （迷路を右手法で進むように）

だから用途が違う

BFSDFS
最短距離求まる求まらない
連結判定できるできる
使うものqueue再帰 or stack
実装やや長い短い

「最短」と書いてあったら BFS。これは覚えてしまっていい。

なぜ BFS で最短が求まるのか

BFS は距離が近い順に訪問する。だから、あるマスに初めて到達したときの距離が、そのマスへの最短距離になる。後から別ルートで来ても、必ずそれ以上の距離になっている。

だから「訪問済みなら無視する」で正しい。これが DFS だと成り立たない（遠回りで先に着く可能性がある）。

訪問済みの記録は push のときに行う。pop のときに記録すると、同じマスが複数回キューに入る。計算量が壊れて TLE の原因になる。

重みがある場合は使えない

BFS が最短を求められるのは、辺の重みがすべて同じ（=1）のときだけ。移動コストが辺ごとに違う場合はダイクストラ法が必要になる。これはD問題以降の話。

動的計画法：考え方の順序

DP は暗記ではなく、決まった手順で組み立てられる。

手順

  1. dp[i] が何を表すかを日本語で決める（ここが9割）
  2. 初期値を決める（dp[0] は何か）
  3. 遷移を書く（dp[i] は何から作られるか）
  4. 答えがどこにあるか確認する

1番が最重要

「dp[i] = i 段目に到達する方法の数」のように、言葉で言い切れるまで詰める。ここが曖昧なまま式を書くと必ず詰まる。

言葉にできたら、遷移は自然に決まる。「i 段目に来られるのは i-1 段目と i-2 段目からだから、dp[i] = dp[i-1] + dp[i-2]」というように。

「もらう DP」と「配る DP」

同じ問題を2通りの向きで書ける。

// もらう: dp[i] を、それより前から集めて決める
for (int i = 1; i <= n; i++) {
    dp[i] = dp[i-1];
    if (i >= 2) dp[i] += dp[i-2];
}

// 配る: dp[i] が確定したら、その先に足しに行く
for (int i = 0; i < n; i++) {
    dp[i+1] += dp[i];
    if (i+2 <= n) dp[i+2] += dp[i];
}

どちらでもいい。「配る」のほうが遷移先に条件があるとき書きやすい（範囲外チェックを1箇所に集められる）。

なぜ DP が速いのか

全探索だと同じ計算を何度も繰り返す。DP は途中結果を保存して使い回すので、指数時間が多項式時間になる。

階段の例なら、素朴な再帰は O(2N) だが、DP なら O(N)。同じ「i 段目までの方法数」を1回しか計算しないから。

ナップサックの dp[i][j] の意味

「i 個目まで見て、重さの合計が j のときの最大価値」。2次元になるのは、状態を決めるのに「どこまで見たか」と「今の重さ」の2つが必要だから。

DP の次元数は「状態を一意に決めるのに必要な情報の数」で決まる。ここを意識すると、初見の DP でも次元が決められる。

貪欲法：いつ正しいのか

貪欲は「その場で最善を選び続ける」解法。速いが、常に正しいとは限らない。

正しくない例

硬貨が 1円、4円、5円のとき、8円を最小枚数で作る。

貪欲（大きい方から）: 5 + 1 + 1 + 1 = 4枚
最適:                4 + 4         = 2枚

日本の硬貨（1, 5, 10, 50, 100, 500）では貪欲が正しいが、硬貨の組み合わせによっては成り立たない。

正しさを確認する方法

  1. 小さいケースを手で解いて、貪欲の答えと一致するか確認
  2. 反例を作ろうとしてみる（作れなければ正しい可能性が高い）
  3. 時間があれば、全探索の解と比較するプログラムを書く

正しい貪欲の典型

問題選び方理由
区間スケジューリング終了時刻が早い順早く終わるほど後に余裕が残る
交互に大きいものを取る降順にソート各手番で最大を取るのが最適
コストの小さい順に処理昇順にソート交換しても改善しないことが示せる

貪欲の多くは「ソートしてから順に見る」形になる。何順にソートするかが解法の核心。開始時刻順・終了時刻順・比の順など、候補を挙げて小さいケースで試す。

しゃくとり法

「条件を満たす連続する区間」を数える／探すときに使う。区間の左端と右端を、両方とも前にしか進めないことで O(N) にする。

使える条件

区間を広げると条件が満たされにくくなり、縮めると満たされやすくなる、という単調性が必要。

例：「和が K 以下の連続部分列の個数」。区間を広げれば和は増える（条件を満たしにくくなる）ので、しゃくとりが使える。

int r = 0;
ll sum = 0, ans = 0;

for (int l = 0; l < n; l++) {
    // r をできるだけ伸ばす
    while (r < n && sum + a[r] <= k) {
        sum += a[r];
        r++;
    }
    ans += r - l;      // l を左端とする区間の個数

    // l を1つ進める前に、その分を引く
    if (l == r) r++;   // 空区間になったら r も進める
    else sum -= a[l];
}
cout << ans << "\n";

なぜ O(N) なのか

二重ループに見えるが、r は全体を通して一度も戻らない。l も r も 0 から N まで一方向に進むだけなので、合計の移動回数は 2N 回。だから O(N)。

l == r の処理を忘れると無限ループする。区間が空になったときに r も進めないと、l が r を追い越してしまう。

鳩の巣原理

N+1 個のものを N 個の箱に入れると、2個以上入る箱が必ずある。

まず、今夜は何も決めない

コンテスト直後の30分は、その日で一番判断が悪くなる時間帯です。

「向いてない」「やめよう」「頭が悪い」「時間の無駄だった」。この種の結論は、疲労と直後のショックが言わせています。同じことを明日の昼に考えると、たいてい違う結論になります。

今日は閉じていいです。判断は明日以降にします。

直後にやること・避けること

やっていい避けたほうがいい
画面を閉じる順位表を上から下まで眺める
水を飲む、何か食べるTLを開いて他人の結果を見る
横になる「やめる」と決める、宣言する

明日読む用に問題をブックマークするその場で全問復習しようとする

「今日の自分について長文を書く」を避ける理由は、直後に書いた文章が後から読むとかなりきついものになるからです。書きたくなったら下書きに置いて、明日読み返してから出すか決めてください。

レートが下がった、の正確な意味

気持ちの話に入る前に、事実を確認します。ここを誤解していると必要以上に落ち込みます。

レートは強さの絶対値ではない

AtCoderのレートは「直近のパフォーマンスの加重平均」です。1回の成績が今のレートより低ければ下がり、高ければ上がる。それだけの仕組みです。

起きたこと本当の意味
レートが下がったその回の成績が、今の自分のレートより低かった
いつもより解けなかったその回に出た典型が、たまたま手持ちになかった
同レートの人が解けていたその典型を先に知っていた。知識の差であって才能の差ではない
時間内に間に合わなかった実装速度の問題。練習量で確実に上がる部分
単純ミスでWAを重ねた手順の問題。チェックリストで減らせる

参加回数が少ないうちは、レートは実力を表していません。AtCoderのレートは初期の数回で大きく振れる設計になっていて、収束するまで十数回かかります。最初の数回の数字は「まだ測定中」です。

1回の結果に含まれるノイズ

その回の成績を左右する要素を並べると、実力以外が意外と多いことが分かります。

  • その回に出た典型を知っていたかどうか
  • 問題文の言い回しが自分に読みやすかったか
  • その日の睡眠時間と体調
  • 実装でつまらないミスをしたかどうか
  • 詰まった問題を切り上げる判断ができたか
  • その回の難易度設定（回によってぶれる）

実力が同じでも、この組み合わせで結果は何段階も変わります。だから1回の結果はほぼノイズで、意味があるのは数十回分の傾向だけです。

レートに反映されないもの

レートは競技上の便宜的な数字であって、人間の能力の総合評価ではありません。次のようなものは一切反映されません。

  • 実務でコードを書く力
  • 設計する力、読みやすく書く力
  • 分からないものを調べて理解する力
  • 粘る力、続ける力
  • 人に説明する力

レートは「その競技のその時点でのスコア」以上のものではありません。ゲームのスコアと同じです。低いことに人格的な意味づけをしないでください。

「解けなかった」を分解する

漠然と「ダメだった」と思っていると対処できません。何が起きたかを分けると、全部が対処可能な問題になります。

A問題で詰まった └ ほぼ問題文の読み違い。実力ではなく読解の事故 └ 対処: 制約と出力形式を先に読む習慣をつける B問題で詰まった └ たいてい実装の詰まり。方針は合っていたのに書けなかった └ 対処: 貼れるコードから始めて書き換える C問題で詰まった └ ほぼ典型を知らなかっただけ └ 対処: 解説を読んで手持ちに1つ足す。次から解ける 方針は分かったのに時間切れ └ 実装速度の問題。練習量で確実に上がる └ 対処: 同じ典型を数問解いて書き慣れる WAを何度も出した └ 手順の問題。ロジックではなく確認不足 └ 対処: 提出前チェックリストを機械的に通す 全部詰まった └ 体調を疑う。眠い日の結果は実力を反映しない └ 対処: 結果を実力の証拠として扱わない

どれも「才能がない」という説明を必要としません。全部、次に変えられる要素です。

C問題が解けないのは頭の問題ではない

C問題で必要なのは、手持ちのパターン数です。典型は有限で、しかも全部公開されています。知れば解けるし、知らなければ何時間考えても解けません。

つまり、C問題が解けない状態と解ける状態を分けているのは、知識の量であって思考力ではありません。これは時間で埋まります。

頭の中で起きていること

結果が悪かったとき、事実より極端な結論が出てくるのには決まった型があります。名前がついているものなので、当てはめると距離が取れます。

出てくる考え何が起きているか事実
全然できなかった 全か無かで判断している 1問でも解けていれば0ではない。部分的にできたことがある
いつも失敗する 1回を全体に広げている 「いつも」の中身を数えると、たいてい2〜3回
みんな解けてるのに できた人だけが目に入っている 解けなかった人は投稿しない。見えていないだけ

これらは意志が弱いから出てくるのではなく、疲れているときに誰の頭でも起きることです。気合いで打ち消すのではなく、「今その型が出ているな」と気づくだけで威力が落ちます。

特に注意したい「べき思考」

「このレートならC問題は解けるべき」「毎日精進すべき」。この形の考えは、達成しても満足が来ず、達成できないと自責だけが残ります。

「べき」が出てきたら、その基準がどこから来たかを探してみてください。たいてい、誰か他人の水準を勝手に自分に適用しているだけです。

  • 疲労が溜まっている
  • 体調を崩している
  • 直前に別のことで消耗している
  • 気分の波が来ている

これらの日に出た結果を自己評価の根拠にすると、実力より低い自己像が固定されます。測定条件が悪かった日のデータは捨てるのが正しい扱いです。

調子が悪い日は、出ないという選択肢もある。1回休んでもレートは変わらない（減らない）。unrated で出る、あるいは過去問だけ触るのでも継続にはなる。

コンテスト後に強く落ち込む日

結果に対して落ち込みが不釣り合いに大きいと感じたら、その日は振り返りをしないでください。振り返りは判断力が要る作業で、落ちている状態でやると事実より悪い結論が出ます。

復習は翌日以降でも間に合います。問題は消えません。

今日の結果から取り出すもの

負けた回のほうが、勝った回より情報量が多いです。ただし全部拾おうとすると、結局何もできません。

復習は1問だけ

  1. 解けなかった問題を1問だけ選ぶ
  2. 解説を読む。完全に理解できなくても最後まで目を通す
  3. 「何を知っていれば解けたか」を一言でメモする
  4. 今日はそこで終わり。実装は明日でいい

全部復習しようとすると、結局復習しなくなります。1問だけと決めると、負けた日でも手が動きます。1回のコンテストで典型が1つ増えるなら、10回出れば10個増えます。それで十分に速いペースです。

メモの書き方

長く書く必要はありません。あとで自分が読んで思い出せればいいので、1行で十分です。

ABC300 C — 「最大値を最小化」は答えを二分探索
ABC301 D — bit全探索。N≤20 を見落としてた
ABC302 C — 実装は分かってたが時間切れ。もっと素直に書く

これを続けると、自分が何で落としているかの傾向が見えてきます。「典型不足」なのか「実装速度」なのか「読み違い」なのか。傾向が分かれば練習の方向が決まります。

明日やること

  1. 今日の1問の解説を読み直す
  2. 同じ典型の問題をもう1問だけ解く
  3. それで終わり

やめたくなったときの整理

やめること自体は悪くありません。他にやりたいことができた、時間が取れなくなった、単純に楽しくない。どれも正当な理由です。

ただ、決めるタイミングだけ間違えないようにします。

やめる以外の選択肢

  • unrated で出る（結果を気にせず参加できる）

どれも継続です。「毎週必ず出る」を守れなくても続けられます。

一度離れて、また戻ってくる人はたくさんいます。やめると宣言する必要も、続けると宣言する必要もありません。やりたくなったらやればいいだけです。

「動ける時に動く」は、必ずしも正解ではない。波がある人ほど、良い日に上限を作っておくほうが総量は増える。

あらかじめ「やらない日があっていい」と決めておくと、実際にやらなかった日に自責が発生しません。自責が発生しないと、翌日再開しやすくなります。

連続記録が途切れると再開できなくなるタイプなら、最初から連続記録を目標にしないでください。

競プロの範囲を超えていると感じたら

ここまでは競技の話です。ただ、落ち込みが競技の範囲を超えていると感じることがあります。

  • 結果に対して落ち込みが不釣り合いに大きい
  • 何日も引きずる
  • 競プロ以外のことにも手がつかなくなる
  • 眠れない、食べられない状態が続く
  • 自分を強く責める言葉が止まらない

これらが当てはまるとき、原因は競プロではありません。競プロがきっかけになっただけで、もともと消耗が溜まっていた可能性が高いです。

そういうときは、競技の問題として扱わないほうがいいです。復習も精進も後回しにして、休むことを優先してください。問題は消えません。

落ちている日の出場

「うまくいかなかった日に」は終わったあとの話。ここは出る前と出ている間の話です。状態に波があるなら、出場そのものを設計の対象にしたほうが、結果として長く続きます。

出る前に決めること

状態が落ちている日に決めるべきなのは「出るか出ないか」ではありません。「Ratedで出るかどうか」です。

Ratedを外せば、出場に伴う損失はゼロになります。解く問題も、かかる時間も、身につくものも変わりません。変わるのはレートに記録されるかどうかだけです。

選択肢は2つではなく4つあります。「出ない」は失敗ではなく、4つのうちの1つです。

選択肢レート向いている状態
Ratedで出る変動する昼まで普通に動けていて、夜も余力が残っている日
Unratedで登録して出る変動しないやりたい気持ちはあるが、21時の状態が読めない日
ADT・バーチャルに振り替える変動しない夜に動けないと分かっている日
今日はやらない変動しない（登録していなければ）何も決められない日

波がある人にとって、上3つの使い分けが競技の中核です。どれを選んでも練習は成立します。

Rated登録の落とし穴

ここは仕組みの話です。誤解していると、出られなかった日に損をします。

Algorithm系のコンテスト（ABC・ARC・AGC）は、Ratedで参加登録した場合、問題を1問も開かず1回も提出しなくてもレーティング計算の対象になります。「とりあえずRatedで登録して、21時の体調で決める」は、波がある人にとって最も避けたい形です。出られなかった日が0完の成績として記録されます。

登録のしかた結果的に出なかった場合
Ratedで登録0完としてレートが下がる
Unratedで登録何も起きない
登録しない何も起きない

登録の取り消しと種別の変更は、コンテスト開始前ならできます。開始後は取り消せません。つまり21時を過ぎた時点で、その日の条件は固定されます。

登録の種別は、前日か、遅くとも夕方までに決めてください。

夜の自分に選ばせないのが要点です。判断力が落ちる時間帯に「Ratedで出るべきか」を考えると、体調の判断ではなく自己評価の話になってしまいます。

迷いが少しでもある日は、Unratedで登録しておくのが安全側です。調子が良ければそのまま全問解けばいいだけで、失うものはレートの上昇分だけです。

21時という時刻の問題

ABCは21:00開始、100分、終わるのは22:40です。夕方以降に状態が落ちるタイプの人にとって、コンテスト時間は1日で一番条件の悪い時間帯に重なっています。

ここで押さえておきたいのは、これは実力の問題ではなく時間割の問題だということです。同じ問題を昼に解けば解ける、という状況は普通に起こります。夜に解けなかったことを実力の測定値として扱うと、実際より低い自己像が残ります。

夜を避けて練習する手段

手段時刻形式レート

バーチャル参加好きな時刻過去のABCを本番と同じ時間で再現変動なし

昼のADTとバーチャルを練習の本体にして、21時のRatedは余力がある日のボーナスにする、という設計にしていいです。

レートは出られる日に出れば後からついてきます。逆に、状態の悪い日に無理に出てレートを削り、その結果で落ち込んで数日動けなくなるほうが、総量では大きく損をします。

当日の判断手順

判断を当日の夜の自分に任せないために、基準は先に決めてメモに置いておきます。

  1. 前日または当日の昼に、登録の種別を決める（Rated / Unrated / 登録しない）
  2. 20:30 になったら、下のチェックだけを見る
  3. 該当が2つ以上あれば、その日はRatedで出ない

20:30のチェック

  • 今日、昼のうちに一度も集中できていない
  • 食事をまともに取れていない
  • 睡眠が明らかに足りていない
  • 生活のリズム（食事・睡眠・服薬など）が崩れている
  • 今日すでに別のことで消耗しきっている
  • 「どうでもいい」「何をやっても無駄」という感覚が続いている
  • 同じ考えが頭の中で繰り返し再生されている
  • 問題文を読む気力がない（解ける自信ではなく、読む気力の話）

このチェックは「出る資格があるか」を測るものではありません。測定条件が整っているかどうかだけを見ています。条件が悪い日に出た結果は、実力のデータとして使えません。使えないデータを取るためにレートを払う必要はない、という判断です。

逆に、該当が0〜1個ならRatedで出て構いません。「調子が悪い気がする」という感覚だけで降りると、出られる日まで削ることになります。感覚ではなく項目で判断するために、このチェックを使ってください。

コンテスト中に崩れたとき

状態が落ちている日にコンテスト中に起きることは、だいたい決まっています。実力が落ちたのではなく、状態による現象です。あらかじめ対処を決めておけば、その場で考えなくて済みます。

起きること対処
文字は読めるのに、問題の意味が頭の中で組み立たない1回で理解できない前提にする。入力欄と出力欄とサンプルだけ先に見る。声に出して読むと入ることがある

A問題を1問出して閉じた日も、参加した日です。

0完で閉じた日も、決めておいた登録の種別を守れたのなら設計通りに動けています。競技としての成績と、その日の運用の成否は別の話です。

撤退ラインを先に決める

コンテスト中に「やめるかどうか」を判断するのは、状態が悪い日には難しい作業です。だから条件は開始前に決めておきます。

条件対応

同じ問題で30分進んでいない次に移らず、その日は終了
文章が目に入らない、同じ行を何度も読んでいる閉じる。読解が落ちている状態で続けても時間だけ減る
手が震える、涙が出る、呼吸が浅くなるその場で閉じる。結果は見ない
「自分には価値がない」という考えに移っている競技の範囲を超えています。閉じて最後の節を読んでください

途中でやめても、提出済みのACは残ります。Unratedなら何も起きません。Ratedでも、1回の成績はレートを構成する加重平均の中のノイズ1つです。1回の撤退でレートが壊れることはありません。

撤退を決めたら、その回の順位表と提出履歴は開かないでください。開いても情報は増えず、判断が悪い時間帯に材料だけが増えます。

終わった直後から数時間

ここは「今夜は何も決めない」と重なる部分があります。状態が落ちている日に特に効くのは次の4つです。

反芻が始まったら復習しない

同じ場面が頭の中で繰り返し再生され始めたら、そこで作業を止めてください。復習は判断力を使う作業です。状態が悪いときにやると、事実より悪い結論が出て、それが記憶として残ります。問題は翌日も消えません。

夜に出た結論は翌日の昼まで保留

「向いていない」「やめたほうがいい」「時間を無駄にした」。この種の結論は、翌日の昼に同じ強さで思うかどうかだけが判断材料になります。夜の時点では判定しません。

書くなら下書きに置く

その日の自分について長文を書きたくなったら、下書きに置いて翌日読み返してから出すか決めてください。撤退や引退の宣言も同じ扱いにします。

記録だけ残す

感想ではなく、条件と結果だけを1行で残します。自己評価のためではなく、次に判断するときの材料にするためです。

項目書くこと
日付・形式9/12 Unrated（またはRated / ADT / バーチャル）
状態良い / 普通 / 悪い の3段階だけ

次にやること1つだけ。「Cの累積和を1問」など

この記録は、状態と結果の相関を後から見るのに使えます。「悪い日の結果が実力ではない」ことを、感覚ではなくデータで確認できるようになります。通院しているなら、波が来ている時期を伝える材料としてもそのまま使えます。

波を前提にした設計

調子が一定な人向けの目標設定をそのまま持ち込むと、波が来た時点で自責が発生し、そこで止まります。最初から前提を変えておきます。

ありがちな目標波がある前提の形
毎週ABCに出る月に1〜2回、状態のいい日にRatedで出る
毎日1時間精進する触れる日に1問開く。解けなくてもいい
レートを○○まで上げる3ヶ月で解ける型を3つ増やす
解けるまで考える20分で解説を見る
連続参加記録を伸ばす記録は数えない

Ratedを0回にする期間を作ってよい

「最初の3ヶ月はRatedに一度も出ない」という進め方は成立します。ADTとバーチャルと過去問だけで、A・B・Cの典型は全部練習できます。レートが動かないだけで、解けるようになったことは全部残ります。

数えるのは「出た回数」ではなく「触った日数」にしてください。

出場は状態に左右されますが、1問開くだけなら悪い日でもできることがあります。触った日数で数えると、波の底でも記録が途切れません。

良い日に詰め込まない

続けるための設計と同じ話です。調子がいい日に一気にやると反動で数日止まり、平均すると遅くなります。良い日にこそ上限を決めておいてください。

競技の調整で対処できない領域

ここまでは、競プロ側で調整できる範囲の話です。次のどれかに当てはまるときは、範囲の外にあります。

  • 落ち込みが、競プロをやっていない時間帯にも続いている
  • 2週間以上、状態が戻らない
  • 睡眠や食事が崩れたままになっている
  • 以前は楽しめたことが、何も楽しめない
  • 自分を責める言葉が止まらない

このとき、出場の設計を変えても状況は動きません。このガイドでできるのは、競プロがその状態をさらに悪化させないようにするところまでです。

やること

  • 競プロの予定を外す。登録していれば、開始前に取り消す
  • 通院しているなら、次の診察で波が来ていることを伝える。記録があればそのまま材料になる
